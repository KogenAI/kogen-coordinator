package main

import (
 "encoding/json"
 "os"

 "github.com/KogenAI/kogen-coordinator/project"
 "kogen-loop/sys"
)

func main() {
 _, _, err := project.CheckIntent(os.Args[1], "demo", nil)
 lines := []string{}
 if err != nil {
  e, ok := sys.AsError(err)
  if !ok { panic(err) }
  lines = e.Lines
 }
 if err := json.NewEncoder(os.Stdout).Encode(lines); err != nil { panic(err) }
}
